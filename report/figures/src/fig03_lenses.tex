% report/figures/src/fig03_lenses.tex
% The seven N3 lenses as a 7-row matrix: lens badge, source construct, the missing element it
% looks for, predicted knowledge type (with tacitness), and the human channel to ask. Content
% from sections/B_lenses.tex (checked against gapmap/src), which agrees with research/n3/README.md
% except where noted in figures/README.md. Columns are left edges on one fixed x list; rows sit
% at one fixed pitch.
\documentclass[tikz,border=1mm]{standalone}
\usepackage{fontspec}
\setsansfont{Source Sans Pro}
\input{figstyle.tex}

\newlength{\rowpitch}\setlength{\rowpitch}{12.5mm}
\newcommand{\colA}{0mm}
\newcommand{\colB}{15.5mm}
\newcommand{\colC}{46.5mm}
\newcommand{\colD}{77.5mm}
\newcommand{\colE}{116.5mm}
\newcommand{\rowy}[1]{-#1*\rowpitch}
\newcommand{\sub}[1]{\\[0.3mm]{\fontsize{7}{8}\selectfont\color{gray700}#1}}

\tikzset{
  cell/.style={anchor=west, align=flush left, inner xsep=0pt, inner ysep=0.5mm},
  hdr/.style={anchor=south west, inner sep=0pt, text=gray700,
              font=\sffamily\fontsize{8}{9.5}\selectfont\scshape},
  badge/.style={procdet, anchor=west, minimum width=13mm, text width=11mm, minimum height=7mm,
                inner sep=0pt, font=\sffamily\bfseries\fontsize{8.5}{10}\selectfont},
  rule/.style={draw=gray200, line width=0.4pt},
}

% #1 row, #2 code, #3 construct, #4 looks for, #5 type, #6 ask via
\newcommand{\lensrow}[6]{%
  \node[badge] at (\colA,\rowy{#1}) {#2};
  \node[cell, text width=30mm] at (\colB,\rowy{#1}) {#3};
  \node[cell, text width=29mm] at (\colC,\rowy{#1}) {#4};
  \node[cell, text width=37mm] at (\colD,\rowy{#1}) {#5};
  \node[cell, text width=31mm] at (\colE,\rowy{#1}) {#6};
}

\begin{document}
\begin{tikzpicture}

  \node[hdr] at (\colA,-0.45*\rowpitch) {Lens};
  \node[hdr] at (\colB,-0.45*\rowpitch) {Source construct};
  \node[hdr] at (\colC,-0.45*\rowpitch) {Looks for};
  \node[hdr] at (\colD,-0.45*\rowpitch) {Predicted type};
  \node[hdr] at (\colE,-0.45*\rowpitch) {Ask via};
  \draw[draw=gray400, line width=0.6pt] (\colA,-0.5*\rowpitch) -- (147.5mm,-0.5*\rowpitch);

  \lensrow{1}{DISC}{CTA (CDM):\\cue as discrimination}{boundary of a\\judgment term}
    {cue\sub{perceptual or collective}}{contrasting cases;\\rated vignettes}
  \lensrow{2}{DIAG}{PARI, ECD:\\rival causes}{a sign that tells\\the causes apart}
    {interpretation, cue\sub{relational}}{CDM on a recalled case, then contrasting cases}
  \lensrow{3}{SEL}{GOMS, HTA, KC:\\selection rule}{situation-to-option\\mapping}
    {decision\sub{relational}}{CDM: options and basis of the choice}
  \lensrow{4}{HEDGE}{CBR, KLI:\\when not to apply}{the exception\\condition}
    {decision\sub{relational or collective}}{CDM hypotheticals; boundary vignettes}
  \lensrow{5}{GUARD}{Automaticity:\\self-check habits}{how the error\\is noticed}
    {check, metacognition\sub{automated}}{observation first;\\probe is weak}
  \lensrow{6}{WHY}{Szulanski:\\causal ambiguity}{the reason\\(rationale)}
    {rationale\sub{relational}}{targeted confirmation question}
  \lensrow{7}{RESULT}{PARI: result\\interpretation}{what a result means\\and where it leads}
    {interpretation, expectancy\sub{relational}}{CDM on a recalled case; process tracing}

  \foreach \r in {1,...,6}
    \draw[rule] (\colA,{-(\r+0.5)*\rowpitch}) -- (147.5mm,{-(\r+0.5)*\rowpitch});
  \draw[draw=gray400, line width=0.6pt] (\colA,-7.5*\rowpitch) -- (147.5mm,-7.5*\rowpitch);

  \node[anchor=north west, inner sep=0pt, text=gray700,
        font=\sffamily\itshape\fontsize{7.5}{9}\selectfont]
    at (\colA,{-7.5*\rowpitch-2mm})
    {The literature supports the constructs in column 2, not the lenses themselves.};

\end{tikzpicture}
\end{document}
